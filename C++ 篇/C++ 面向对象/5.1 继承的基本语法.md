# 5.1 继承的基本语法 <a id="5-1"></a> <a id="5"></a>

[← 返回目录](<C++面向对象编程.md>)

定义一些类时，下级成员除了拥有上一级的共性，还有自己的特性。使用继承，减少重复代码

- 继承可以减少重复代码

- **语法：`class 子类: 继承方式 父类`**

- 子类：派生类

- 父类：基类

```cpp
class BasePage{
public:
    void header(){
        cout << "首页、公开课、登录、注册...（公共头部）" << endl;
    }
    void footer(){
        cout << "帮助中心、交流合作、站内地图...（公共底部）" << endl;
    }
    void left(){
        cout << "Java、Python、C++...（公共分类列表）" << endl;
    }
};

class Java: public BasePage{
public:
    void content(){
        cout << "Java学科视频" << endl;
    }
};

class Python: public BasePage{
public:
    void content(){
        cout << "Python学科视频" << endl;
    }
};

class CPP: public BasePage{
public:
    void content(){
        cout << "C++学科视频" << endl;
    }
};

void edge(){
    cout << "----------------------------------------" << endl;
}

void test(){
    Java ja;
    ja.header();
    ja.left();
    ja.content();
    ja.footer();
    edge();

    Python py;
    py.header();
    py.left();
    py.content();
    py.footer();
    edge();

    CPP cpp;
    cpp.header();
    cpp.left();
    cpp.content();
    cpp.footer();
    edge();
}
```
