## 2.2.1 传统方式初始化

```cpp
class Person{
public:
    int age;
    char gender;
    int height;

    Person(int a, char g, int h){
        age = a;
        gender = g;
        height = h;
    }

    void show(){
        cout << "年龄:" << age << " 性别:" << gender << " 身高:" << height << endl;
    }
};

int main(){
    Person p(20, 'M', 175);
    p.show();

    return 0;
}
```
