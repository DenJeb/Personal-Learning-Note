# 3. algorithm：对数据操作 <a id="algorithm"></a>

[← 返回目录](<C++ STL.md>)

## 3.1 简介

algorithm 头文件提供了一系列通用算法，它们通过迭代器区间 `[begin, end)` 操作容器，不直接依赖容器类型。

> 💡 binary_search / lower_bound / upper_bound 要求区间已有序（原理见[折半查找](<../../数据结构/7. 查找.md#7-3>)）；merge 需先给输出容器预留足够容量（如 c.resize(a.size() + b.size())）。各排序算法见[第 8 章](<../../数据结构/8. 内部排序.md#ch8>)。

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <functional>
#include <utility>
using namespace std;

bool cmp(int a, int b){
    return a > b;
}

int main(){
    vector<int> v = {3, 1, 2, 2, 3};

    // 默认升序
    sort(v.begin(), v.end());                    // {1, 2, 2, 3, 3}

    // 降序
    sort(v.begin(), v.end(), greater<int>());    // {3, 3, 2, 2, 1}

    // 自定义排序：比较函数
    sort(v.begin(), v.end(), cmp);               // 降序

    // 自定义排序：lambda 表达式
    sort(v.begin(), v.end(), [](int a, int b){
        return a > b;
    });

    // 反转
    reverse(v.begin(), v.end());

    // 去重：先排序，让相同元素相邻
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());   // {1, 2, 3}

    // 二分查找：区间必须有序，返回是否存在
    cout << binary_search(v.begin(), v.end(), 2) << endl;   // 1

    // 下界 / 上界：区间必须有序，返回迭代器
    cout << *lower_bound(v.begin(), v.end(), 2) << endl;    // 第一个 ≥ 2：2
    cout << *upper_bound(v.begin(), v.end(), 2) << endl;    // 第一个 > 2：3

    // 极值：返回迭代器
    cout << *max_element(v.begin(), v.end()) << endl;   // 最大值：3
    cout << *min_element(v.begin(), v.end()) << endl;   // 最小值：1

    // 查找：返回迭代器，找不到返回 v.end()
    cout << *find(v.begin(), v.end(), 2) << endl;   // 2

    // 计数
    cout << count(v.begin(), v.end(), 2) << endl;   // 1

    // 全排列：先升序，再用 do-while 枚举所有排列
    sort(v.begin(), v.end());
    do {
        for (int x : v)
            cout << x << " ";
        cout << endl;
    } while (next_permutation(v.begin(), v.end()));

    // 求和：初始值 0（需要 <numeric> 头文件）
    cout << accumulate(v.begin(), v.end(), 0) << endl;   // 6

    // 填充：全部填充为 0
    fill(v.begin(), v.end(), 0);

    // 交换
    vector<int> v1 = {1, 2};
    vector<int> v2 = {3, 4};
    swap(v1, v2);

    // 合并：c 需预留足够容量
    vector<int> a = {1, 3};
    vector<int> b = {2, 4};
    vector<int> c(a.size() + b.size());
    merge(a.begin(), a.end(), b.begin(), b.end(), c.begin());

    return 0;
}
```
